     5	  status: number;
     6	  title: string;
     7	  detail?: string;

     9	}
    10	
    11	export interface ProblemFieldError {
    12	  field: string | null;
    13	  message: string;
    14	}
    15	
    16	export function problemDetailMessage(error: unknown): string {
    17	  if (error instanceof HttpErrorResponse) {
    18	    const problem = error.error as ProblemDetail | null;
    19	    return problem?.detail ?? problem?.title ?? error.message;
    20	  }
    21	  return 'Unexpected error';
    22	}
    23	
    24	export function loadErrorMessage(error: unknown, notFoundMessage: string): string {
    25	  if (error instanceof HttpErrorResponse && error.status === 404) {
    26	    return notFoundMessage;
    27	  }
    28	  return problemDetailMessage(error);
    29	}
    30	
    31	export function problemDetailToFormErrors<T extends object>(
    32	  error: unknown,
    33	  form: FieldTree<T>,
    34	): ValidationError.WithOptionalFieldTree[] {
    35	  const fieldErrors =
    36	    error instanceof HttpErrorResponse ? (error.error as ProblemDetail | null)?.errors : undefined;
    37	  if (!fieldErrors?.length) {
    38	    // A 409 conflict carries no field errors: it is an error of the whole form, not of a field.
    39	    return [{ kind: 'server', message: problemDetailMessage(error), fieldTree: form }];
    40	  }
    41	  const fields = form as unknown as Record<string, FieldTree<unknown> | undefined>;
    42	  return fieldErrors.map(({ field, message }) => ({
    43	    kind: 'server',
    44	    message,
    45	    fieldTree: (field === null ? undefined : fields[field]) ?? form,
    46	  }));
    47	}

    26	  return {

    37	  return {

    80	          const id = this.id();

    82	            const saved = await firstValueFrom(
